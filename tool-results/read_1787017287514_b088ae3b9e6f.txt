    82→interface EditFormState {
    83→  name: string;
    84→  tagline: string;
    85→  description: string;

    87→  province: string;
    88→  city: string;

    94→  businessSize: string;
    95→  yearFounded: string;
    96→  employeeCount: string;
    97→  annualRevenue: string;
    98→  cipcNumber: string;
    99→  bbbeeLevel: string;
   100→  services: string[];
   101→  products: string[];

   103→
   104→function editFormFromBusiness(b: Business): EditFormState {
   105→  return {
   106→    name: b.name ?? "",
   107→    tagline: b.tagline ?? "",
   108→    description: b.description ?? "",
   109→    industryId: b.industryId ?? "",
   110→    province: b.province ?? "",
   111→    city: b.city ?? "",
   112→    address: b.address ?? "",
   113→    website: b.website ?? "",
   114→    email: b.email ?? "",
   115→    phone: b.phone ?? "",
   116→    whatsapp: b.whatsapp ?? "",
   117→    businessSize: b.businessSize ?? "",
   118→    yearFounded: b.yearFounded != null ? String(b.yearFounded) : "",
   119→    employeeCount: b.employeeCount ?? "",
   120→    annualRevenue: b.annualRevenue ?? "",
   121→    cipcNumber: b.cipcNumber ?? "",
   122→    bbbeeLevel: b.bbbeeLevel ?? "",
   123→    services: (b.services ?? []).map((s) => s.name),
   124→    products: (b.products ?? []).map((p) => p.name),
   125→  };
   126→}
   127→
   128→function buildEditPayload(form: EditFormState) {
   129→  return {
   130→    name: form.name.trim(),
   131→    tagline: form.tagline.trim() || undefined,
   132→    description: form.description.trim() || undefined,
   133→    industryId: form.industryId || undefined,
   134→    province: form.province || undefined,
   135→    city: form.city.trim() || undefined,
   136→    address: form.address.trim() || undefined,
   137→    website: form.website.trim() || undefined,
   138→    email: form.email.trim() || undefined,
   139→    phone: form.phone.trim() || undefined,
   140→    whatsapp: form.whatsapp.trim() || undefined,
   141→    businessSize: form.businessSize || undefined,
   142→    yearFounded: form.yearFounded ? Number(form.yearFounded) : undefined,
   143→    employeeCount: form.employeeCount || undefined,
   144→    annualRevenue: form.annualRevenue || undefined,
   145→    cipcNumber: form.cipcNumber.trim() || undefined,
   146→    bbbeeLevel: form.bbbeeLevel || undefined,
   147→    services: form.services,
   148→    products: form.products,
   149→  };
   150→}

   152→// Best-effort local merge of edited fields back onto the business object
   153→// (used only if the PATCH response doesn't include a fresh business payload).
   154→function editBusinessMerge(form: EditFormState): Partial<Business> {
   155→  return {
   156→    name: form.name.trim(),
   157→    tagline: form.tagline.trim() || null,
   158→    description: form.description.trim() || null,
   159→    industryId: form.industryId || null,
   160→    province: form.province || null,
   161→    city: form.city.trim() || null,
   162→    address: form.address.trim() || null,
   163→    website: form.website.trim() || null,
   164→    email: form.email.trim() || null,
   165→    phone: form.phone.trim() || null,
   166→    whatsapp: form.whatsapp.trim() || null,
   167→    businessSize: form.businessSize || null,
   168→    yearFounded: form.yearFounded ? Number(form.yearFounded) : null,
   169→    employeeCount: form.employeeCount || null,
   170→    annualRevenue: form.annualRevenue || null,
   171→    cipcNumber: form.cipcNumber.trim() || null,
   172→    bbbeeLevel: form.bbbeeLevel || null,
   173→    services: form.services.map((name, i) => ({ id: `local-${i}-${name}`, name })),
   174→    products: form.products.map((name, i) => ({ id: `local-${i}-${name}`, name })),
   175→  };
   176→}

   178→export function BusinessDetailView() {
   179→  const route = useApp((s) => s.route);
   180→  if (route.name !== "dashboard-business") return null;
   181→  // `key` ensures the inner component fully remounts on id change → all
   182→  // state (loading, notFound, verify form) resets cleanly between businesses.
   183→  return <BusinessDetail key={route.id} id={route.id} />;
   184→}
   185→
   186→function BusinessDetail({ id }: { id: string }) {
   187→  const navigate = useApp((s) => s.navigate);
   188→  const { toast } = useToast();
   189→
   190→  const [business, setBusiness] = useState<Business | null>(null);
   191→  const [loading, setLoading] = useState(true);
   192→  const [notFound, setNotFound] = useState(false);
   193→  const [showVerifyForm, setShowVerifyForm] = useState(false);
   194→
   195→  // edit dialog state
   196→  const [editOpen, setEditOpen] = useState(false);
   197→  const [editForm, setEditForm] = useState<EditFormState | null>(null);
   198→  const [industries, setIndustries] = useState<Industry[]>([]);
   199→  const [saving, setSaving] = useState(false);
   200→
   201→  // verification form state
   202→  const [verificationType, setVerificationType] = useState("CIPC");
   203→  const [documentUrl, setDocumentUrl] = useState("");
   204→  const [notes, setNotes] = useState("");
   205→  const [submitting, setSubmitting] = useState(false);
   206→
   207→  // Load industries (for the edit dialog's industry Select).
   208→  useEffect(() => {
   209→    let cancelled = false;
   210→    api<{ industries: Industry[] }>("/api/industries")
   211→      .then((d) => {
   212→        if (cancelled) return;
   213→        setIndustries(d.industries ?? []);
   214→      })
   215→      .catch(() => {
   216→        /* silent — Select just stays empty */
   217→      });
   218→    return () => {
   219→      cancelled = true;
   220→    };
   221→  }, []);
   222→
   223→  useEffect(() => {
   224→    let cancelled = false;
   225→    api<{ items: Business[] }>("/api/businesses/owner")
   226→      .then((d) => {
   227→        if (cancelled) return;
   228→        const found = (d.items ?? []).find((b) => b.id === id);
   229→        if (found) setBusiness(found);
   230→        else setNotFound(true);
   231→      })
   232→      .catch(() => {
   233→        if (cancelled) return;
   234→        setNotFound(true);
   235→      })
   236→      .finally(() => {
   237→        if (!cancelled) setLoading(false);
   238→      });
   239→    return () => {
   240→      cancelled = true;
   241→    };
   242→  }, [id]);
   243→
   244→  const refetch = () => {
   245→    api<{ items: Business[] }>("/api/businesses/owner")
   246→      .then((d) => {
   247→        const found = (d.items ?? []).find((b) => b.id === id);
   248→        if (found) setBusiness(found);
   249→      })
   250→      .catch(() => {
   251→        /* silent */
   252→      });
   253→  };
   254→
   255→  const openEdit = () => {
   256→    if (!business) return;
   257→    setEditForm(editFormFromBusiness(business));
   258→    setEditOpen(true);
   259→  };
   260→
   261→  const handleEditSave = async () => {
   262→    if (!business || !editForm) return;
   263→    if (!editForm.name.trim()) {
   264→      toast({ title: "Business name is required.", variant: "destructive" });
   265→      return;
   266→    }
   267→    setSaving(true);
   268→    try {
   269→      const payload = buildEditPayload(editForm);
   270→      const res = await api<{ business: Business }>(
   271→        `/api/businesses/${encodeURIComponent(business.id)}/edit`,
   272→        { method: "PATCH", json: payload },
   273→      );
   274→      setBusiness(res.business ?? { ...business, ...editBusinessMerge(editForm) });
   275→      setEditOpen(false);
   276→      setEditForm(null);
   277→      toast({ title: "Business updated", description: "Your changes are live." });
   278→      refetch();
   279→    } catch (err) {
   280→      const message = err instanceof Error ? err.message : "Could not save changes.";
   281→      toast({
   282→        title: "Couldn't save changes",
   283→        description: message,
   284→        variant: "destructive",
   285→      });
   286→    } finally {
   287→      setSaving(false);
   288→    }
   289→  };
   290→
   291→  const handleVerifySubmit = async (e: FormEvent) => {
   292→    e.preventDefault();
   293→    if (!business) return;
   294→    setSubmitting(true);
   295→    try {
   296→      await api("/api/verification", {
   297→        method: "POST",
   298→        json: {
   299→          businessId: business.id,
   300→          verificationType,
   301→          notes: notes.trim() || undefined,
   302→          documentUrl: documentUrl.trim() || undefined,
   303→        },
   304→      });
   305→      toast({
   306→        title: "Verification submitted",
   307→        description: "We'll review within 2 business days.",
   308→      });
   309→      setShowVerifyForm(false);
   310→      setNotes("");
   311→      setDocumentUrl("");
   312→      refetch();
   313→    } catch (err) {
   314→      const message = err instanceof Error ? err.message : "Could not submit verification.";
   315→      toast({
   316→        title: "Couldn't submit verification",
   317→        description: message,
   318→        variant: "destructive",
   319→      });
   320→    } finally {
   321→      setSubmitting(false);
   322→    }
   323→  };
   324→
   325→  if (loading) {
   326→    return (
   327→      <div className="space-y-6">
   328→        <Skeleton className="h-8 w-40 rounded-md" />
   329→        <Skeleton className="h-36 w-full rounded-2xl" />
   330→        <div className="grid gap-6 lg:grid-cols-3">
   331→          <Skeleton className="lg:col-span-2 h-72 rounded-xl" />
   332→          <Skeleton className="h-72 rounded-xl" />
   333→        </div>
   334→      </div>
   335→    );
   336→  }
   337→
   338→  if (notFound || !business) {
   339→    return (
   340→      <EmptyState
   341→        icon={Building2}
   342→        title="Business not found"
   343→        description="This business doesn't exist, or you don't have access to manage it."
   344→        action={
   345→          <Button onClick={() => navigate({ name: "dashboard-businesses" })} className="bg-ink text-cream hover:bg-ink/90">
   346→            <ArrowLeft className="mr-1.5 h-4 w-4" /> Back to my businesses
   347→          </Button>
   348→        }
   349→      />
   350→    );
   351→  }
   352→
   353→  const status = business.verificationStatus;
   354→  const isVerified = status === "VERIFIED";
   355→  const isPending = status === "PENDING";
   356→
   357→  return (
   358→    <div className="space-y-6">
   359→      <button
   360→        onClick={() => navigate({ name: "dashboard-businesses" })}
   361→        className="inline-flex items-center gap-1.5 text-sm text-muted-foreground hover:text-ink"
   362→      >
   363→        <ArrowLeft className="h-4 w-4" /> Back to my businesses
   364→      </button>
   365→
   366→      {/* Header card */}
   367→      <div className="rounded-2xl border border-border bg-card p-6 sm:p-8">
   368→        <div className="flex flex-col gap-5 sm:flex-row sm:items-start">
   369→          {business.logoUrl ? (
   370→            <img
   371→              src={business.logoUrl}
   372→              alt={business.name}
   373→              className="h-16 w-16 shrink-0 rounded-xl object-cover"
   374→            />
   375→          ) : (
   376→            <div className="flex h-16 w-16 shrink-0 items-center justify-center rounded-xl bg-ink font-display text-2xl text-cream">
   377→              {business.name[0]?.toUpperCase() ?? "?"}
   378→            </div>
   379→          )}
   380→
   381→          <div className="min-w-0 flex-1">
   382→            <div className="flex flex-wrap items-center gap-2">
   383→              <h1 className="font-display text-2xl tracking-tight text-ink sm:text-3xl">
   384→                {business.name}
   385→              </h1>
   386→              <VerifiedBadge status={status} size="md" />
   387→              <BBBEEBadge level={business.bbbeeLevel} />
   388→              {business.featured && <Pill tone="cream">Featured</Pill>}
   389→            </div>
   390→
   391→            {business.tagline && (
   392→              <p className="mt-1 text-sm text-muted-foreground">{business.tagline}</p>
   393→            )}
   394→
   395→            <div className="mt-3 flex flex-wrap items-center gap-3 text-[11px] text-muted-foreground">
   396→              <span className="inline-flex items-center gap-1">
   397→                <MapPin className="h-3 w-3" /> {provinceCity(business)}
   398→              </span>
   399→              {business.industry && (
   400→                <span className="inline-flex items-center gap-1">
   401→                  <Building2 className="h-3 w-3" /> {business.industry.name}
   402→                </span>
   403→              )}
   404→              <span className="inline-flex items-center gap-1">
   405→                <Eye className="h-3 w-3" /> {formatNumber(business.views)} views
   406→              </span>
   407→              <span className="inline-flex items-center gap-1">
   408→                <Calendar className="h-3 w-3" /> Added {formatDate(business.createdAt, { day: "numeric", month: "short", year: "numeric" })}
   409→              </span>
   410→            </div>
   411→
   412→            {/* Profile completion */}
   413→            <div className="mt-4 flex items-center gap-3">
   414→              <div className="h-1.5 w-40 max-w-full">
   415→                <Progress value={business.profileCompletion} className="h-1.5" />
   416→              </div>
   417→              <span className="text-[11px] text-muted-foreground">
   418→                {business.profileCompletion}% complete
   419→              </span>
   420→              {business.profileCompletion < 100 && (
   421→                <span className="hidden text-[11px] text-foreground/40 sm:inline">
   422→                  · add more detail to reach 100%
   423→                </span>
   424→              )}
   425→            </div>
   426→          </div>
   427→        </div>
   428→
   429→        <Separator className="my-6" />
   430→
   431→        <div className="flex flex-wrap gap-2">
   432→          <Button
   433→            onClick={() => navigate({ name: "business", slug: business.slug })}
   434→            className="bg-ink text-cream hover:bg-ink/90"
   435→          >
   436→            View public profile <ArrowRight className="ml-1 h-4 w-4" />
   437→          </Button>
   438→          <Button variant="outline" onClick={openEdit}>
   439→            <Pencil className="mr-1.5 h-4 w-4" /> Edit details
   440→          </Button>
   441→          {!isVerified && !isPending && (
   442→            <Button
   443→              variant="outline"
   444→              onClick={() => setShowVerifyForm((v) => !v)}
   445→              className="border-sage/30 bg-sage/10 text-sage hover:bg-sage/20 hover:text-sage"
   446→            >
   447→              <ShieldCheck className="mr-1.5 h-4 w-4" /> Submit for verification
   448→            </Button>
   449→          )}
   450→        </div>
   451→      </div>
   452→
   453→      {/* Verification inline form */}
   454→      {showVerifyForm && !isVerified && !isPending && (
   455→        <div className="rounded-2xl border border-sage/30 bg-sage/5 p-6 sm:p-8">
   456→          <div className="flex items-start justify-between gap-4">
   457→            <div>
   458→              <h2 className="font-display text-xl tracking-tight text-ink">Request verification</h2>
   459→              <p className="mt-1 text-sm text-muted-foreground">
   460→                Choose the verification type you want to submit. We'll review within 2 business days.
   461→              </p>
   462→            </div>
   463→            <button
   464→              onClick={() => setShowVerifyForm(false)}
   465→              className="inline-flex h-8 w-8 items-center justify-center rounded-full text-muted-foreground hover:bg-foreground/10 hover:text-foreground"
   466→              aria-label="Close form"
   467→            >
   468→              <X className="h-4 w-4" />
   469→            </button>
   470→          </div>
   471→
   472→          <form onSubmit={handleVerifySubmit} className="mt-5 space-y-5">
   473→            <div className="space-y-1.5">
   474→              <Label>Verification type</Label>
   475→              <Select value={verificationType} onValueChange={setVerificationType}>
   476→                <SelectTrigger className="w-full">
   477→                  <SelectValue />
   478→                </SelectTrigger>
   479→                <SelectContent>
   480→                  {VERIFICATION_TYPES.map((t) => (
   481→                    <SelectItem key={t.value} value={t.value}>
   482→                      {t.label}
   483→                    </SelectItem>
   484→                  ))}
   485→                </SelectContent>
   486→              </Select>
   487→            </div>
   488→
   489→            <div className="space-y-1.5">
   490→              <Label htmlFor="documentUrl">Document URL (optional)</Label>
   491→              <Input
   492→                id="documentUrl"
   493→                value={documentUrl}
   494→                onChange={(e) => setDocumentUrl(e.target.value)}
   495→                placeholder="File upload coming soon — paste a link for now"
   496→                inputMode="url"
   497→              />
   498→              <p className="inline-flex items-center gap-1 text-[11px] text-muted-foreground">
   499→                <AlertCircle className="h-3 w-3" /> File uploads arrive shortly — paste a public link to your document for now.
   500→              </p>
   501→            </div>
   502→
   503→            <div className="space-y-1.5">
   504→              <Label htmlFor="notes">Notes (optional)</Label>
   505→              <Textarea
   506→                id="notes"
   507→                value={notes}
   508→                onChange={(e) => setNotes(e.target.value)}
   509→                rows={3}
   510→                placeholder="Anything you'd like the reviewer to know — registration numbers, references, etc."
   511→              />
   512→            </div>
   513→
   514→            <div className="flex justify-end gap-2">
   515→              <Button type="button" variant="outline" onClick={() => setShowVerifyForm(false)}>
   516→                Cancel
   517→              </Button>
   518→              <Button type="submit" disabled={submitting} className="bg-ink text-cream hover:bg-ink/90">
   519→                {submitting ? "Submitting…" : "Submit request"}
   520→              </Button>
   521→            </div>
   522→          </form>
   523→        </div>
   524→      )}
   525→
   526→      {/* Body */}
   527→      <div className="grid gap-6 lg:grid-cols-3">
   528→        {/* Main column */}
   529→        <div className="space-y-6 lg:col-span-2">
   530→          <StatusCard status={status} onVerifyClick={() => setShowVerifyForm(true)} />
   531→
   532→          {/* Services & products card */}
   533→          <div className="rounded-xl border border-border bg-card">
   534→            <div className="border-b border-border px-5 py-4">
   535→              <h2 className="font-display text-lg tracking-tight">Services & products</h2>
   536→              <p className="text-sm text-muted-foreground">What your business offers.</p>
   537→            </div>
   538→            <div className="space-y-5 p-5">
   539→              <div>
   540→                <div className="mb-2 flex items-center gap-1.5 text-[11px] uppercase tracking-wide text-muted-foreground">
   541→                  <Briefcase className="h-3 w-3" /> Services
   542→                </div>
   543→                {(business.services ?? []).length === 0 ? (
   544→                  <p className="text-sm text-muted-foreground">No services listed yet.</p>
   545→                ) : (
   546→                  <div className="flex flex-wrap gap-1.5">
