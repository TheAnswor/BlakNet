    63→    63→export function BusinessProfileView() {

    70→    70→

    83→    83→  const [reviewText, setReviewText] = useState("");

    97→    97→          method: "POST",

    99→    99→          /* noop */
   100→   100→        });

   104→   104→        if (err.status === 404 || err.status === 400) {
   105→   105→          setNotFound(true);
   106→   106→        } else {
   107→   107→          setNotFound(true);
   108→   108→        }
   109→   109→      })
   110→   110→      .finally(() => {
   111→   111→        if (!cancelled) setLoading(false);

   132→   132→        `/api/businesses/${encodeURIComponent(slug)}/reviews`,
   133→   133→        { method: "POST", json: { rating: reviewRating, review: reviewText.trim(), reviewerCompany: reviewCompany.trim() || undefined } },
   134→   134→      );

   136→   136→      setReviewText("");
   137→   137→      setReviewCompany("");
   138→   138→      setReviewRating(5);
   139→   139→      setShowReviewForm(false);
   140→   140→      toast({ title: "Review posted", description: "Thanks for sharing your experience." });
   141→   141→    } catch (err) {
   142→   142→      const msg = err instanceof Error ? err.message : "Could not submit review.";
   143→   143→      toast({ title: "Could not submit review", description: msg });
   144→   144→    } finally {
   145→   145→      setSubmittingReview(false);
   146→   146→    }

   155→   155→        await navigator.clipboard.writeText(url);
   156→   156→        toast({ title: "Link copied", description: "Share this business with your network." });
   157→   157→      }
   158→   158→    } catch {
   159→   159→      /* user dismissed share — noop */
   160→   160→    }
   161→   161→  };

   205→   205→            type="button"
   206→   206→            onClick={() => navigate({ name: "directory" })}

   743→   743→    [

   748→   748→    ] as const
