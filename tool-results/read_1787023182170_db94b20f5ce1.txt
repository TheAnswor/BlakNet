   100→  bbbeeLevel: string;

   102→  coverUrl: string | null;
   103→  services: string[];

   106→
   107→function editFormFromBusiness(b: Business): EditFormState {
   108→  return {
   109→    name: b.name ?? "",
   110→    tagline: b.tagline ?? "",
   111→    description: b.description ?? "",
   112→    industryId: b.industryId ?? "",
   113→    province: b.province ?? "",
   114→    city: b.city ?? "",
   115→    address: b.address ?? "",
   116→    website: b.website ?? "",
   117→    email: b.email ?? "",
   118→    phone: b.phone ?? "",
   119→    whatsapp: b.whatsapp ?? "",
   120→    businessSize: b.businessSize ?? "",
   121→    yearFounded: b.yearFounded != null ? String(b.yearFounded) : "",
   122→    employeeCount: b.employeeCount ?? "",
   123→    annualRevenue: b.annualRevenue ?? "",
   124→    cipcNumber: b.cipcNumber ?? "",
   125→    bbbeeLevel: b.bbbeeLevel ?? "",
   126→    logoUrl: b.logoUrl ?? null,
   127→    coverUrl: b.coverUrl ?? null,
   128→    services: (b.services ?? []).map((s) => s.name),
   129→    products: (b.products ?? []).map((p) => p.name),
   130→  };
   131→}

   133→function buildEditPayload(form: EditFormState) {
   134→  return {
   135→    name: form.name.trim(),
   136→    tagline: form.tagline.trim() || undefined,
   137→    description: form.description.trim() || undefined,
   138→    industryId: form.industryId || undefined,
   139→    province: form.province || undefined,
   140→    city: form.city.trim() || undefined,
   141→    address: form.address.trim() || undefined,
   142→    website: form.website.trim() || undefined,
   143→    email: form.email.trim() || undefined,
   144→    phone: form.phone.trim() || undefined,
   145→    whatsapp: form.whatsapp.trim() || undefined,
   146→    businessSize: form.businessSize || undefined,
   147→    yearFounded: form.yearFounded ? Number(form.yearFounded) : undefined,
   148→    employeeCount: form.employeeCount || undefined,
   149→    annualRevenue: form.annualRevenue || undefined,
   150→    cipcNumber: form.cipcNumber.trim() || undefined,
   151→    bbbeeLevel: form.bbbeeLevel || undefined,
   152→    logoUrl: form.logoUrl,
   153→    coverUrl: form.coverUrl,
   154→    services: form.services,
   155→    products: form.products,
   156→  };
   157→}

   159→// Best-effort local merge of edited fields back onto the business object
   160→// (used only if the PATCH response doesn't include a fresh business payload).
   161→function editBusinessMerge(form: EditFormState): Partial<Business> {
   162→  return {
   163→    name: form.name.trim(),
   164→    tagline: form.tagline.trim() || null,
   165→    description: form.description.trim() || null,
   166→    industryId: form.industryId || null,
   167→    province: form.province || null,
   168→    city: form.city.trim() || null,
   169→    address: form.address.trim() || null,
   170→    website: form.website.trim() || null,
   171→    email: form.email.trim() || null,
   172→    phone: form.phone.trim() || null,
   173→    whatsapp: form.whatsapp.trim() || null,
   174→    businessSize: form.businessSize || null,
   175→    yearFounded: form.yearFounded ? Number(form.yearFounded) : null,
   176→    employeeCount: form.employeeCount || null,
   177→    annualRevenue: form.annualRevenue || null,
   178→    cipcNumber: form.cipcNumber.trim() || null,
   179→    bbbeeLevel: form.bbbeeLevel || null,
   180→    logoUrl: form.logoUrl,
   181→    coverUrl: form.coverUrl,
   182→    services: form.services.map((name, i) => ({ id: `local-${i}-${name}`, name })),
   183→    products: form.products.map((name, i) => ({ id: `local-${i}-${name}`, name })),
   184→  };
   185→}

   187→export function BusinessDetailView() {
   188→  const route = useApp((s) => s.route);
   189→  if (route.name !== "dashboard-business") return null;
   190→  // `key` ensures the inner component fully remounts on id change → all
   191→  // state (loading, notFound, verify form) resets cleanly between businesses.
   192→  return <BusinessDetail key={route.id} id={route.id} />;
   193→}
   194→
   195→function BusinessDetail({ id }: { id: string }) {
   196→  const navigate = useApp((s) => s.navigate);
   197→  const { toast } = useToast();
   198→
   199→  const [business, setBusiness] = useState<Business | null>(null);
   200→  const [loading, setLoading] = useState(true);
   201→  const [notFound, setNotFound] = useState(false);
   202→  const [showVerifyForm, setShowVerifyForm] = useState(false);
   203→
   204→  // edit dialog state
   205→  const [editOpen, setEditOpen] = useState(false);
   206→  const [editForm, setEditForm] = useState<EditFormState | null>(null);
   207→  const [industries, setIndustries] = useState<Industry[]>([]);
   208→  const [saving, setSaving] = useState(false);
   209→
   210→  // verification form state
   211→  const [verificationType, setVerificationType] = useState("CIPC");
   212→  const [documentUrl, setDocumentUrl] = useState("");
   213→  const [notes, setNotes] = useState("");
   214→  const [submitting, setSubmitting] = useState(false);
   215→
   216→  // Load industries (for the edit dialog's industry Select).
   217→  useEffect(() => {
   218→    let cancelled = false;
   219→    api<{ industries: Industry[] }>("/api/industries")
   220→      .then((d) => {
   221→        if (cancelled) return;
   222→        setIndustries(d.industries ?? []);
   223→      })
   224→      .catch(() => {
   225→        /* silent — Select just stays empty */
   226→      });
   227→    return () => {
   228→      cancelled = true;
   229→    };
   230→  }, []);
   231→
   232→  useEffect(() => {
   233→    let cancelled = false;
   234→    api<{ items: Business[] }>("/api/businesses/owner")
   235→      .then((d) => {
   236→        if (cancelled) return;
   237→        const found = (d.items ?? []).find((b) => b.id === id);
   238→        if (found) setBusiness(found);
   239→        else setNotFound(true);
   240→      })
   241→      .catch(() => {
   242→        if (cancelled) return;
   243→        setNotFound(true);
   244→      })
   245→      .finally(() => {
   246→        if (!cancelled) setLoading(false);
   247→      });
   248→    return () => {
   249→      cancelled = true;
   250→    };
   251→  }, [id]);
   252→
   253→  const refetch = () => {
   254→    api<{ items: Business[] }>("/api/businesses/owner")
   255→      .then((d) => {
   256→        const found = (d.items ?? []).find((b) => b.id === id);
   257→        if (found) setBusiness(found);
   258→      })
   259→      .catch(() => {
   260→        /* silent */
   261→      });
   262→  };
   263→
   264→  const openEdit = () => {
   265→    if (!business) return;
   266→    setEditForm(editFormFromBusiness(business));
   267→    setEditOpen(true);
   268→  };
   269→
   270→  const handleEditSave = async () => {
   271→    if (!business || !editForm) return;
   272→    if (!editForm.name.trim()) {
   273→      toast({ title: "Business name is required.", variant: "destructive" });
   274→      return;
   275→    }
   276→    setSaving(true);
   277→    try {
   278→      const payload = buildEditPayload(editForm);
   279→      const res = await api<{ business: Business }>(
   280→        `/api/businesses/${encodeURIComponent(business.id)}/edit`,
   281→        { method: "PATCH", json: payload },
   282→      );
   283→      setBusiness(res.business ?? { ...business, ...editBusinessMerge(editForm) });
   284→      setEditOpen(false);
   285→      setEditForm(null);
   286→      toast({ title: "Business updated", description: "Your changes are live." });
   287→      refetch();
   288→    } catch (err) {
   289→      const message = err instanceof Error ? err.message : "Could not save changes.";
   290→      toast({
   291→        title: "Couldn't save changes",
   292→        description: message,
   293→        variant: "destructive",
   294→      });
   295→    } finally {
   296→      setSaving(false);
   297→    }
   298→  };
   299→
   300→  const handleVerifySubmit = async (e: FormEvent) => {
   301→    e.preventDefault();
   302→    if (!business) return;
   303→    setSubmitting(true);
   304→    try {
   305→      await api("/api/verification", {
   306→        method: "POST",
   307→        json: {
   308→          businessId: business.id,
   309→          verificationType,
   310→          notes: notes.trim() || undefined,
   311→          documentUrl: documentUrl.trim() || undefined,
   312→        },
   313→      });
   314→      toast({
   315→        title: "Verification submitted",
   316→        description: "We'll review within 2 business days.",
   317→      });
   318→      setShowVerifyForm(false);
   319→      setNotes("");
   320→      setDocumentUrl("");
   321→      refetch();
   322→    } catch (err) {
   323→      const message = err instanceof Error ? err.message : "Could not submit verification.";
   324→      toast({
   325→        title: "Couldn't submit verification",
   326→        description: message,
   327→        variant: "destructive",
   328→      });
   329→    } finally {
   330→      setSubmitting(false);
   331→    }
   332→  };
   333→
   334→  if (loading) {
   335→    return (
   336→      <div className="space-y-6">
   337→        <Skeleton className="h-8 w-40 rounded-md" />
   338→        <Skeleton className="h-36 w-full rounded-2xl" />
   339→        <div className="grid gap-6 lg:grid-cols-3">
   340→          <Skeleton className="lg:col-span-2 h-72 rounded-xl" />
   341→          <Skeleton className="h-72 rounded-xl" />
   342→        </div>
   343→      </div>
   344→    );
   345→  }
   346→
   347→  if (notFound || !business) {
   348→    return (
   349→      <EmptyState
   350→        icon={Building2}
   351→        title="Business not found"
   352→        description="This business doesn't exist, or you don't have access to manage it."
   353→        action={
   354→          <Button onClick={() => navigate({ name: "dashboard-businesses" })} className="bg-ink text-cream hover:bg-ink/90">
   355→            <ArrowLeft className="mr-1.5 h-4 w-4" /> Back to my businesses
   356→          </Button>
   357→        }
   358→      />
   359→    );
   360→  }
   361→
   362→  const status = business.verificationStatus;
   363→  const isVerified = status === "VERIFIED";
   364→  const isPending = status === "PENDING";
   365→
   366→  return (
   367→    <div className="space-y-6">
   368→      <button
   369→        onClick={() => navigate({ name: "dashboard-businesses" })}
   370→        className="inline-flex items-center gap-1.5 text-sm text-muted-foreground hover:text-ink"
   371→      >
   372→        <ArrowLeft className="h-4 w-4" /> Back to my businesses
   373→      </button>
   374→
   375→      {/* Header card */}
   376→      <div className="rounded-2xl border border-border bg-card p-6 sm:p-8">
   377→        <div className="flex flex-col gap-5 sm:flex-row sm:items-start">
   378→          {business.logoUrl ? (
   379→            <img
   380→              src={business.logoUrl}
   381→              alt={business.name}
   382→              className="h-16 w-16 shrink-0 rounded-xl object-cover"
   383→            />
   384→          ) : (
   385→            <div className="flex h-16 w-16 shrink-0 items-center justify-center rounded-xl bg-ink font-display text-2xl text-cream">
   386→              {business.name[0]?.toUpperCase() ?? "?"}
   387→            </div>
   388→          )}
   389→
   390→          <div className="min-w-0 flex-1">
   391→            <div className="flex flex-wrap items-center gap-2">
   392→              <h1 className="font-display text-2xl tracking-tight text-ink sm:text-3xl">
   393→                {business.name}
   394→              </h1>
   395→              <VerifiedBadge status={status} size="md" />
   396→              <BBBEEBadge level={business.bbbeeLevel} />
   397→              {business.featured && <Pill tone="cream">Featured</Pill>}
   398→            </div>
   399→
   400→            {business.tagline && (
   401→              <p className="mt-1 text-sm text-muted-foreground">{business.tagline}</p>
   402→            )}
   403→
   404→            <div className="mt-3 flex flex-wrap items-center gap-3 text-[11px] text-muted-foreground">
   405→              <span className="inline-flex items-center gap-1">
   406→                <MapPin className="h-3 w-3" /> {provinceCity(business)}
   407→              </span>
   408→              {business.industry && (
   409→                <span className="inline-flex items-center gap-1">
   410→                  <Building2 className="h-3 w-3" /> {business.industry.name}
   411→                </span>
   412→              )}
   413→              <span className="inline-flex items-center gap-1">
   414→                <Eye className="h-3 w-3" /> {formatNumber(business.views)} views
   415→              </span>
   416→              <span className="inline-flex items-center gap-1">
   417→                <Calendar className="h-3 w-3" /> Added {formatDate(business.createdAt, { day: "numeric", month: "short", year: "numeric" })}
   418→              </span>
   419→            </div>
   420→
   421→            {/* Profile completion */}
   422→            <div className="mt-4 flex items-center gap-3">
   423→              <div className="h-1.5 w-40 max-w-full">
   424→                <Progress value={business.profileCompletion} className="h-1.5" />
   425→              </div>
   426→              <span className="text-[11px] text-muted-foreground">
   427→                {business.profileCompletion}% complete
   428→              </span>
   429→              {business.profileCompletion < 100 && (
   430→                <span className="hidden text-[11px] text-foreground/40 sm:inline">
   431→                  · add more detail to reach 100%
   432→                </span>
   433→              )}
   434→            </div>
   435→          </div>
   436→        </div>
   437→
   438→        <Separator className="my-6" />
   439→
   440→        <div className="flex flex-wrap gap-2">
   441→          <Button
   442→            onClick={() => navigate({ name: "business", slug: business.slug })}
   443→            className="bg-ink text-cream hover:bg-ink/90"
   444→          >
   445→            View public profile <ArrowRight className="ml-1 h-4 w-4" />
   446→          </Button>
   447→          <Button variant="outline" onClick={openEdit}>
   448→            <Pencil className="mr-1.5 h-4 w-4" /> Edit details
   449→          </Button>
   450→          {!isVerified && !isPending && (
   451→            <Button
   452→              variant="outline"
   453→              onClick={() => setShowVerifyForm((v) => !v)}
   454→              className="border-sage/30 bg-sage/10 text-sage hover:bg-sage/20 hover:text-sage"
   455→            >
   456→              <ShieldCheck className="mr-1.5 h-4 w-4" /> Submit for verification
   457→            </Button>
   458→          )}
   459→        </div>
   460→      </div>
   461→
   462→      {/* Verification inline form */}
   463→      {showVerifyForm && !isVerified && !isPending && (
   464→        <div className="rounded-2xl border border-sage/30 bg-sage/5 p-6 sm:p-8">
   465→          <div className="flex items-start justify-between gap-4">
   466→            <div>
   467→              <h2 className="font-display text-xl tracking-tight text-ink">Request verification</h2>
   468→              <p className="mt-1 text-sm text-muted-foreground">
   469→                Choose the verification type you want to submit. We'll review within 2 business days.
   470→              </p>
   471→            </div>
   472→            <button
   473→              onClick={() => setShowVerifyForm(false)}
   474→              className="inline-flex h-8 w-8 items-center justify-center rounded-full text-muted-foreground hover:bg-foreground/10 hover:text-foreground"
   475→              aria-label="Close form"
   476→            >
   477→              <X className="h-4 w-4" />
   478→            </button>
   479→          </div>
   480→
   481→          <form onSubmit={handleVerifySubmit} className="mt-5 space-y-5">
   482→            <div className="space-y-1.5">
   483→              <Label>Verification type</Label>
   484→              <Select value={verificationType} onValueChange={setVerificationType}>
   485→                <SelectTrigger className="w-full">
   486→                  <SelectValue />
   487→                </SelectTrigger>
   488→                <SelectContent>
   489→                  {VERIFICATION_TYPES.map((t) => (
   490→                    <SelectItem key={t.value} value={t.value}>
   491→                      {t.label}
   492→                    </SelectItem>
   493→                  ))}
   494→                </SelectContent>
   495→              </Select>
   496→            </div>
   497→
   498→            <div className="space-y-1.5">
   499→              <Label htmlFor="documentUrl">Document URL (optional)</Label>
   500→              <Input
   501→                id="documentUrl"
   502→                value={documentUrl}
   503→                onChange={(e) => setDocumentUrl(e.target.value)}
   504→                placeholder="File upload coming soon — paste a link for now"
   505→                inputMode="url"
   506→              />
   507→              <p className="inline-flex items-center gap-1 text-[11px] text-muted-foreground">
   508→                <AlertCircle className="h-3 w-3" /> File uploads arrive shortly — paste a public link to your document for now.
   509→              </p>
   510→            </div>
   511→
   512→            <div className="space-y-1.5">
   513→              <Label htmlFor="notes">Notes (optional)</Label>
   514→              <Textarea
   515→                id="notes"
   516→                value={notes}
   517→                onChange={(e) => setNotes(e.target.value)}
   518→                rows={3}
   519→                placeholder="Anything you'd like the reviewer to know — registration numbers, references, etc."
   520→              />
   521→            </div>
   522→
   523→            <div className="flex justify-end gap-2">
   524→              <Button type="button" variant="outline" onClick={() => setShowVerifyForm(false)}>
   525→                Cancel
   526→              </Button>
   527→              <Button type="submit" disabled={submitting} className="bg-ink text-cream hover:bg-ink/90">
   528→                {submitting ? "Submitting…" : "Submit request"}
   529→              </Button>
   530→            </div>
   531→          </form>
   532→        </div>
   533→      )}
   534→
   535→      {/* Body */}
   536→      <div className="grid gap-6 lg:grid-cols-3">
   537→        {/* Main column */}
   538→        <div className="space-y-6 lg:col-span-2">
   539→          <StatusCard status={status} onVerifyClick={() => setShowVerifyForm(true)} />
   540→
   541→          {/* Services & products card */}
   542→          <div className="rounded-xl border border-border bg-card">
   543→            <div className="border-b border-border px-5 py-4">
   544→              <h2 className="font-display text-lg tracking-tight">Services & products</h2>
   545→              <p className="text-sm text-muted-foreground">What your business offers.</p>
   546→            </div>
   547→            <div className="space-y-5 p-5">
   548→              <div>
   549→                <div className="mb-2 flex items-center gap-1.5 text-[11px] uppercase tracking-wide text-muted-foreground">
   550→                  <Briefcase className="h-3 w-3" /> Services
   551→                </div>
   552→                {(business.services ?? []).length === 0 ? (
   553→                  <p className="text-sm text-muted-foreground">No services listed yet.</p>
   554→                ) : (
   555→                  <div className="flex flex-wrap gap-1.5">
   556→                    {(business.services ?? []).map((s) => (
   557→                      <Pill key={s.id}>{s.name}</Pill>
   558→                    ))}
   559→                  </div>
   560→                )}
   561→              </div>
   562→
   563→              <Separator />
   564→
   565→              <div>
   566→                <div className="mb-2 flex items-center gap-1.5 text-[11px] uppercase tracking-wide text-muted-foreground">
   567→                  <Package className="h-3 w-3" /> Products

  1183→function EditTagInput({
  1184→  label,
  1185→  icon: Icon,
  1186→  placeholder,
  1187→  values,
  1188→  onChange,
  1189→  suggestions,
  1190→}: {
  1191→  label: string;
  1192→  icon: React.ComponentType<{ className?: string }>;
  1193→  placeholder: string;
  1194→  values: string[];
  1195→  onChange: (vals: string[]) => void;

  1199→
  1200→  const add = (raw: string) => {
  1201→    const v = raw.trim();

  1203→    if (values.some((x) => x.toLowerCase() === v.toLowerCase())) {
  1204→      setInput("");

  1211→  const remove = (idx: number) => {

  1220→      remove(values.length - 1);
  1221→    }
  1222→  };
  1223→
  1224→  const remainingSuggestions = suggestions.filter(
  1225→    (s) => !values.some((v) => v.toLowerCase() === s.toLowerCase()),
  1226→  );
  1227→
  1228→  return (
  1229→    <div className="space-y-2">
  1230→      <Label className="inline-flex items-center gap-1.5">

  1233→
  1234→      <div className="flex gap-2">

  1236→          value={input}

  1238→          onKeyDown={handleKeyDown}

  1240→        />

  1260→                type="button"
  1261→                onClick={() => remove(i)}

  1263→                aria-label={`Remove ${v}`}
