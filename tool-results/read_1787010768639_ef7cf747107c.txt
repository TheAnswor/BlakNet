    49→// ---------- types ----------
    50→type ProfileBusiness = Business & {
    51→  rating: number;
    52→  reviewCount: number;
    53→  postCount: number;
    54→  eventCount: number;
    55→  isOwner: boolean;

    57→
    58→interface ProfileResponse {
    59→  business: ProfileBusiness;
    60→}
    61→
    62→interface ReviewPayload {

    65→
    66→const SEARCH_KEY = "blaknet:directory-search";
    67→
    68→export function BusinessProfileView() {
    69→  const route = useApp((s) => s.route);
    70→  if (route.name !== "business") return null;
    71→  // `key` ensures the inner component fully remounts on slug change → all
    72→  // state (loading, notFound, reviews, form) resets cleanly between profiles.
    73→  return <BusinessProfile key={route.slug} slug={route.slug} />;
    74→}
    75→
    76→function BusinessProfile({ slug }: { slug: string }) {
    77→  const navigate = useApp((s) => s.navigate);
    78→  const authUser = useApp((s) => s.authUser);
    79→  const { toast } = useToast();
    80→
    81→  const [business, setBusiness] = useState<ProfileBusiness | null>(null);
    82→  const [loading, setLoading] = useState(true);
    83→  const [notFound, setNotFound] = useState(false);
    84→
    85→  // review form
    86→  const [showReviewForm, setShowReviewForm] = useState(false);
    87→  const [reviewRating, setReviewRating] = useState(5);
    88→  const [reviewText, setReviewText] = useState("");
    89→  const [reviewCompany, setReviewCompany] = useState("");
    90→  const [submittingReview, setSubmittingReview] = useState(false);
    91→  const [localReviews, setLocalReviews] = useState<BusinessReview[] | null>(null);
    92→
    93→  // ---------- fetch business + fire view increment ----------
    94→  useEffect(() => {
    95→    let cancelled = false;
    96→    api<ProfileResponse>(`/api/businesses/${encodeURIComponent(slug)}`)
    97→      .then((d) => {
    98→        if (cancelled) return;
    99→        setBusiness(d.business);
   100→        // fire-and-forget view increment
   101→        fetch(`/api/businesses/${encodeURIComponent(slug)}/view`, {
   102→          method: "POST",
   103→        }).catch(() => {
   104→          /* noop */
   105→        });
   106→      })
   107→      .catch((err: Error & { status?: number }) => {
   108→        if (cancelled) return;
   109→        if (err.status === 404 || err.status === 400) {
   110→          setNotFound(true);
   111→        } else {
   112→          setNotFound(true);
   113→        }
   114→      })
   115→      .finally(() => {
   116→        if (!cancelled) setLoading(false);
   117→      });
   118→    return () => {
   119→      cancelled = true;
   120→    };
   121→  }, [slug]);
   122→
   123→  // ---------- handlers ----------
   124→  const reviews: BusinessReview[] = useMemo(() => {
   125→    if (localReviews) return localReviews;
   126→    return business?.reviews ?? [];
   127→  }, [business, localReviews]);
   128→
   129→  const submitReview = async () => {
   130→    if (!reviewText.trim()) {
   131→      toast({ title: "Review text is required", description: "Please share a few words about your experience." });
   132→      return;
   133→    }
   134→    setSubmittingReview(true);
   135→    try {
   136→      const data = await api<ReviewPayload>(
   137→        `/api/businesses/${encodeURIComponent(slug)}/reviews`,
   138→        { method: "POST", json: { rating: reviewRating, review: reviewText.trim(), reviewerCompany: reviewCompany.trim() || undefined } },
   139→      );
   140→      setLocalReviews((prev) => [data.review, ...(prev ?? business?.reviews ?? [])]);
   141→      setReviewText("");
   142→      setReviewCompany("");
   143→      setReviewRating(5);
   144→      setShowReviewForm(false);
   145→      toast({ title: "Review posted", description: "Thanks for sharing your experience." });
   146→    } catch (err) {
   147→      const msg = err instanceof Error ? err.message : "Could not submit review.";
   148→      toast({ title: "Could not submit review", description: msg });
   149→    } finally {
   150→      setSubmittingReview(false);
   151→    }
   152→  };
   153→
   154→  const shareProfile = async () => {
   155→    const url = typeof window !== "undefined" ? `${window.location.origin}/#/business/${slug}` : "";
   156→    try {
   157→      if (navigator.share) {
   158→        await navigator.share({ title: business?.name ?? "BlakNet business", url });
   159→      } else if (navigator.clipboard) {
   160→        await navigator.clipboard.writeText(url);
   161→        toast({ title: "Link copied", description: "Share this business with your network." });
   162→      }
   163→    } catch {
   164→      /* user dismissed share — noop */
   165→    }
   166→  };
   167→
   168→  const searchByTag = (term: string) => {
   169→    if (typeof window !== "undefined") {
   170→      window.sessionStorage.setItem(SEARCH_KEY, term);
   171→    }
   172→    navigate({ name: "directory" });
   173→  };
   174→
   175→  // ---------- not found ----------
   176→  if (notFound && !loading) {
   177→    return (
   178→      <div className="mx-auto max-w-3xl px-4 py-24">
   179→        <EmptyState
   180→          icon={Frown}
   181→          title="Business not found"
   182→          description="The business you're looking for may have been removed or the link is incorrect."
   183→          action={
   184→            <Button variant="outline" onClick={() => navigate({ name: "directory" })}>
   185→              <ArrowLeft className="mr-1.5 h-4 w-4" /> Back to directory
   186→            </Button>
   187→          }
   188→        />
   189→      </div>
   190→    );
   191→  }
   192→
   193→  // ---------- loading skeleton ----------
   194→  if (loading || !business) {
   195→    return <ProfileSkeleton />;
   196→  }
   197→
   198→  const b = business;
   199→  const yearFounded = b.yearFounded;
   200→  const yearsInBusiness = yearFounded ? Math.max(0, new Date().getFullYear() - yearFounded) : null;
   201→  const websiteHref = b.website ? (b.website.startsWith("http") ? b.website : `https://${b.website}`) : null;
   202→  const whatsappHref = b.whatsapp ? `https://wa.me/${b.whatsapp.replace(/[^0-9]/g, "")}` : null;
   203→
   204→  return (
   205→    <div className="flex flex-col">
   206→      {/* ===== HEADER BAND ===== */}
   207→      <section className="bg-ink-grain text-cream">
   208→        <div className="mx-auto max-w-7xl px-4 py-10 sm:px-6 lg:px-8 lg:py-14">
   209→          <button
   210→            type="button"
   211→            onClick={() => navigate({ name: "directory" })}
   212→            className="mb-6 inline-flex items-center gap-1.5 text-sm text-cream/60 transition-colors hover:text-cream"
   213→          >
   214→            <ArrowLeft className="h-4 w-4" /> Directory
   215→          </button>
   216→
   217→          <div className="flex flex-col gap-6 lg:flex-row lg:items-start lg:justify-between">
   218→            <div className="flex flex-col gap-5 sm:flex-row sm:items-start">
   219→              {/* logo */}
   220→              <div className="h-20 w-20 shrink-0 overflow-hidden rounded-xl border border-cream/15 bg-cream/5">
   221→                {b.logoUrl ? (
   222→                  <img src={b.logoUrl} alt={b.name} className="h-full w-full object-cover" />
   223→                ) : (
   224→                  <div className="flex h-full w-full items-center justify-center font-display text-3xl text-cream">
   225→                    {b.name[0]}
   226→                  </div>
   227→                )}
   228→              </div>
   229→
   230→              <div className="min-w-0">
   231→                <h1 className="font-display text-3xl leading-tight tracking-tight sm:text-4xl md:text-[2.6rem]">
   232→                  {b.name}
   233→                </h1>
   234→                {b.tagline && (
   235→                  <p className="mt-1.5 max-w-xl text-cream/70 sm:text-lg">{b.tagline}</p>
   236→                )}
   237→
   238→                <div className="mt-4 flex flex-wrap items-center gap-2">
   239→                  {b.verificationStatus === "VERIFIED" && <VerifiedBadge status="VERIFIED" size="md" />}
   240→                  {b.verificationStatus === "PENDING" && <VerifiedBadge status="PENDING" size="md" />}
   241→                  {b.bbbeeLevel && <BBBEEBadge level={b.bbbeeLevel} />}
   242→                  {b.featured && <Pill tone="cream">Featured</Pill>}
   243→                  {b.industry?.name && (
   244→                    <Pill tone="cream">
   245→                      <Building2 className="h-3 w-3" />
   246→                      {b.industry.name}
   247→                    </Pill>
   248→                  )}
   249→                  {(b.city || b.province) && (
   250→                    <Pill tone="cream">
   251→                      <MapPin className="h-3 w-3" />
   252→                      {provinceCity(b)}
   253→                    </Pill>
   254→                  )}
   255→                </div>
   256→              </div>
   257→            </div>
   258→
   259→            {/* CTA stack (desktop) */}
   260→            <div className="flex flex-col gap-2 lg:items-end">
   261→              <div className="flex flex-wrap gap-2 lg:justify-end">
   262→                {websiteHref && (
   263→                  <Button asChild className="btn-lift h-10 bg-cream px-4 text-ink shadow-lg hover:bg-cream/90">
   264→                    <a href={websiteHref} target="_blank" rel="noreferrer noopener">
   265→                      <Globe className="h-4 w-4" /> Website
   266→                      <ExternalLink className="h-3 w-3 opacity-60" />
   267→                    </a>
   268→                  </Button>
   269→                )}
   270→                {b.email && (
   271→                  <Button asChild variant="outline" className="h-10 border-cream/25 bg-transparent px-4 text-cream hover:bg-cream/10 hover:text-cream">
   272→                    <a href={`mailto:${b.email}`}>
   273→                      <Mail className="h-4 w-4" /> Email
   274→                    </a>
   275→                  </Button>
   276→                )}
   277→                {whatsappHref && (
   278→                  <Button asChild className={cn(
   279→                    "btn-lift h-10 px-4",
   280→                    websiteHref
   281→                      ? "border border-sage/40 bg-sage/10 text-cream hover:bg-sage/20"
   282→                      : "bg-sage text-cream shadow-lg hover:bg-sage/90",
   283→                  )}>
   284→                    <a href={whatsappHref} target="_blank" rel="noreferrer noopener">
   285→                      <MessageCircle className="h-4 w-4" /> WhatsApp
   286→                    </a>
   287→                  </Button>
   288→                )}
   289→                <Button variant="ghost" onClick={shareProfile} className="h-10 px-4 text-cream/70 hover:bg-cream/10 hover:text-cream">
   290→                  <Share2 className="h-4 w-4" />
   291→                  <span className="hidden sm:inline">Share</span>
   292→                </Button>
   293→              </div>
   294→            </div>
   295→          </div>
   296→        </div>
   297→      </section>
   298→
   299→      {/* ===== PROFILE COMPLETION BAR (owner only) ===== */}
   300→      {b.isOwner && (
   301→        <section className="border-b border-border bg-cream-grain">
   302→          <div className="mx-auto max-w-7xl px-4 py-4 sm:px-6 lg:px-8">
   303→            <div className="flex flex-col gap-3 rounded-xl border border-border bg-card p-4 sm:flex-row sm:items-center sm:justify-between">
   304→              <div className="flex-1">
   305→                <div className="flex items-center justify-between gap-3">
   306→                  <p className="text-sm font-medium">
   307→                    Your profile is {b.profileCompletion}% complete
   308→                  </p>
   309→                  <span className="text-xs text-muted-foreground">
   310→                    {b.profileCompletion >= 80 ? "Looking great!" : "Add more details to rank higher in search."}
   311→                  </span>
   312→                </div>
   313→                <Progress value={b.profileCompletion} className="mt-2 h-1.5" />
   314→              </div>
   315→              <Button
   316→                variant="outline"
   317→                size="sm"
   318→                onClick={() => navigate({ name: "dashboard-business", id: b.id })}
   319→              >
   320→                Edit business
   321→              </Button>
   322→            </div>
   323→          </div>
   324→        </section>
   325→      )}
   326→
   327→      {/* ===== BODY ===== */}
   328→      <section className="bg-background">
   329→        <div className="mx-auto max-w-7xl px-4 pb-10 pt-2 sm:px-6 sm:pt-4 lg:px-8 lg:pb-10">
   330→          <div className="grid gap-8 lg:grid-cols-[minmax(0,2fr)_minmax(0,1fr)]">
   331→            {/* ===== MAIN COLUMN ===== */}
   332→            <div className="flex flex-col gap-6">
   333→              {/* About — first content card, overlaps the dark header for a layered "rising out of the dark" effect */}
   334→              <Card className="-mt-4 sm:-mt-8 relative z-10 shadow-xl">
   335→                <CardTitle>About</CardTitle>
   336→                <Separator className="mb-4" />
   337→                {b.description ? (
   338→                  <div className="space-y-3 text-sm leading-relaxed text-foreground/80">
   339→                    {b.description.split(/\n{2,}|\r\n{2,}/).map((para, i) => (
   340→                      <p key={i}>{para}</p>
   341→                    ))}
   342→                  </div>
   343→                ) : (
   344→                  <p className="text-sm italic text-muted-foreground">
   345→                    This business hasn't added a description yet.
   346→                  </p>
   347→                )}
   348→              </Card>
   349→
   350→              {/* Services & Products — single card with two clearly-titled subsections; pills are clickable (navigate to directory search) */}
   351→              {((b.services && b.services.length > 0) || (b.products && b.products.length > 0)) && (
   352→                <Card>
   353→                  <CardTitle>Services & Products</CardTitle>
   354→                  <Separator className="mb-4" />
   355→
   356→                  {b.services && b.services.length > 0 && (
   357→                    <div className={b.products && b.products.length > 0 ? "mb-6" : ""}>
   358→                      <h3 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   359→                        Services
   360→                      </h3>
   361→                      <div className="flex flex-wrap gap-2">
   362→                        {b.services.map((s) => (
   363→                          <button
   364→                            key={s.id}
   365→                            type="button"
   366→                            onClick={() => searchByTag(s.name)}
   367→                            className="transition-transform hover:-translate-y-0.5"
   368→                          >
   369→                            <Pill
   370→                              tone="neutral"
   371→                              className="cursor-pointer transition-colors hover:bg-ink hover:text-cream hover:border-ink"
   372→                            >
   373→                              {s.name}
   374→                            </Pill>
   375→                          </button>
   376→                        ))}
   377→                      </div>
   378→                    </div>
   379→                  )}
   380→
   381→                  {b.products && b.products.length > 0 && (
   382→                    <div>
   383→                      <h3 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   384→                        Products
   385→                      </h3>
   386→                      <div className="flex flex-wrap gap-2">
   387→                        {b.products.map((p) => (
   388→                          <button
   389→                            key={p.id}
   390→                            type="button"
   391→                            onClick={() => searchByTag(p.name)}
   392→                            className="transition-transform hover:-translate-y-0.5"
   393→                          >
   394→                            <Pill
   395→                              tone="sage"
   396→                              className="cursor-pointer transition-colors hover:bg-ink hover:text-cream hover:border-ink"
   397→                            >
   398→                              {p.name}
   399→                            </Pill>
   400→                          </button>
   401→                        ))}
   402→                      </div>
   403→                    </div>
   404→                  )}
   405→                </Card>
   406→              )}
   407→
   408→              {/* Reviews */}
   409→              <Card>
   410→                {/* Header: rating summary + Leave-a-review CTA */}
   411→                <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   412→                  <div className="flex items-center gap-4">
   413→                    <div className="font-display text-4xl leading-none text-ink">
   414→                      {b.rating > 0 ? b.rating.toFixed(1) : "—"}
   415→                    </div>
   416→                    <div className="flex flex-col">
   417→                      <span className="mb-0.5 text-[11px] font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   418→                        Reviews
   419→                      </span>
   420→                      <StarRating rating={b.rating} showValue={false} />
   421→                      <span className="mt-0.5 text-xs text-muted-foreground">
   422→                        ({formatNumber(b.reviewCount)} {b.reviewCount === 1 ? "review" : "reviews"})
   423→                      </span>
   424→                    </div>
   425→                  </div>
   426→                  {!showReviewForm && (
   427→                    <Button
   428→                      variant="outline"
   429→                      size="sm"
   430→                      onClick={() => {
   431→                        if (!authUser) {
   432→                          navigate({ name: "login" });
   433→                          return;
   434→                        }
   435→                        setShowReviewForm(true);
   436→                      }}
   437→                    >
   438→                      <Plus className="h-4 w-4" /> Leave a review
   439→                    </Button>
   440→                  )}
   441→                </div>
   442→                <Separator className="mb-4 mt-4" />
   443→
   444→                {/* inline review form */}
   445→                {showReviewForm && (
   446→                  <div className="mb-5 rounded-lg border border-border bg-cream-grain/40 p-4">
   447→                    <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center">
   448→                      <label className="text-sm font-medium">Your rating</label>
   449→                      <Select value={String(reviewRating)} onValueChange={(v) => setReviewRating(Number(v))}>
   450→                        <SelectTrigger className="h-9 w-[160px]">
   451→                          <SelectValue />
   452→                        </SelectTrigger>
   453→                        <SelectContent>
   454→                          {[5, 4, 3, 2, 1].map((n) => (
   455→                            <SelectItem key={n} value={String(n)}>
   456→                              {n} {n === 1 ? "star" : "stars"}
   457→                            </SelectItem>
   458→                          ))}
   459→                        </SelectContent>
   460→                      </Select>
   461→                      <div className="flex items-center">
   462→                        <StarRating rating={reviewRating} />
   463→                      </div>
   464→                    </div>
   465→                    <Textarea
   466→                      value={reviewText}
   467→                      onChange={(e) => setReviewText(e.target.value)}
   468→                      placeholder="Share your experience working with this business…"
   469→                      className="min-h-24"
   470→                    />
   471→                    <div className="mt-2">
   472→                      <Input
   473→                        value={reviewCompany}
   474→                        onChange={(e) => setReviewCompany(e.target.value)}
   475→                        placeholder="Your company (optional)"
   476→                        className="h-9"
   477→                      />
   478→                    </div>
   479→                    <div className="mt-3 flex items-center gap-2">
   480→                      <Button onClick={submitReview} disabled={submittingReview} size="sm">
   481→                        {submittingReview ? "Posting…" : "Post review"}
   482→                      </Button>
   483→                      <Button
   484→                        variant="ghost"
   485→                        size="sm"
   486→                        onClick={() => setShowReviewForm(false)}
   487→                        disabled={submittingReview}
   488→                      >
   489→                        Cancel
   490→                      </Button>
   491→                    </div>
   492→                  </div>
   493→                )}
   494→
   495→                {/* review list */}
   496→                {reviews.length === 0 ? (
   497→                  <EmptyState
   498→                    title="No reviews yet"
   499→                    description="Be the first to share your experience with this business."
   500→                    className="border-transparent bg-transparent py-8"
   501→                  />
   502→                ) : (
   503→                  <ul className="flex flex-col gap-4">
   504→                    {reviews.slice(0, 12).map((r) => (
   505→                      <li
   506→                        key={r.id}
   507→                        className="rounded-lg border border-border border-l-2 border-l-sage bg-card/60 p-4"
   508→                      >
   509→                        <div className="flex items-start justify-between gap-3">
   510→                          <div>
   511→                            <p className="text-sm font-medium text-foreground">{r.reviewerName}</p>
   512→                            {r.reviewerCompany && (
   513→                              <p className="text-xs text-muted-foreground">{r.reviewerCompany}</p>
   514→                            )}
   515→                          </div>
   516→                          <span className="text-xs text-muted-foreground">{timeAgo(r.createdAt)}</span>
   517→                        </div>
   518→                        <div className="mt-2">
   519→                          <StarRating rating={r.rating} size={12} />
   520→                        </div>
   521→                        <p className="mt-2 text-sm leading-relaxed text-foreground/80">{r.review}</p>
   522→                      </li>
   523→                    ))}
   524→                  </ul>
   525→                )}
   526→              </Card>
   527→            </div>
   528→
   529→            {/* ===== SIDEBAR ===== */}
   530→            <aside className="flex flex-col gap-6">
   531→              <div className="sticky top-24 flex flex-col gap-6">
   532→                {/* Business Information */}
   533→                <Card>
   534→                  <CardTitle>Business Information</CardTitle>
   535→                  <Separator className="mb-4" />
   536→                  <dl className="flex flex-col gap-3 text-sm">
   537→                    <InfoRow icon={<Calendar className="h-3.5 w-3.5" />} label="Year founded" value={yearFounded != null ? String(yearFounded) : null} />

   771→function sizeLabel(size: string | null): string | null {
   772→  if (!size) return null;
   773→  const found = (
   774→    [
   775→      { value: "micro", label: "Micro (1–10)" },
   776→      { value: "small", label: "Small (11–50)" },
   777→      { value: "medium", label: "Medium (51–200)" },
   778→      { value: "large", label: "Large (200+)" },
   779→    ] as const
   780→  ).find((s) => s.value === size);
   781→  return found ? found.label : size.charAt(0).toUpperCase() + size.slice(1);
   782→}

   784→function prettyDomain(url: string): string {
   785→  try {
   786→    const u = new URL(url);
   787→    return u.hostname.replace(/^www\./, "");
   788→  } catch {
   789→    return url;
   790→  }
   791→}
