   260→   260→            Add a business to BlakNet.

   738→   738→function TagInput({
   739→   739→  label,
   740→   740→  icon: Icon,
   741→   741→  placeholder,
   742→   742→  values,
   743→   743→  onChange,
   744→   744→  suggestions,
   745→   745→}: {
   746→   746→  label: string;
   747→   747→  icon: React.ComponentType<{ className?: string }>;
   748→   748→  placeholder: string;
   749→   749→  values: string[];
   750→   750→  onChange: (vals: string[]) => void;
   751→   751→  suggestions: string[];
   752→   752→}) {
   753→   753→  const [input, setInput] = useState("");
   754→   754→
   755→   755→  const add = (raw: string) => {
   756→   756→    const v = raw.trim();
   757→   757→    if (!v) return;
   758→   758→    if (values.some((x) => x.toLowerCase() === v.toLowerCase())) {
   759→   759→      setInput("");
   760→   760→      return;
   761→   761→    }
   762→   762→    onChange([...values, v]);
   763→   763→    setInput("");
   764→   764→  };
   765→   765→
   766→   766→  const remove = (idx: number) => {
   767→   767→    onChange(values.filter((_, i) => i !== idx));
   768→   768→  };
   769→   769→
   770→   770→  const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
   771→   771→    if (e.key === "Enter" || e.key === ",") {
   772→   772→      e.preventDefault();
   773→   773→      add(input);
   774→   774→    } else if (e.key === "Backspace" && !input && values.length) {
   775→   775→      remove(values.length - 1);
   776→   776→    }
   777→   777→  };
   778→   778→
   779→   779→  const remainingSuggestions = suggestions.filter(
   780→   780→    (s) => !values.some((v) => v.toLowerCase() === s.toLowerCase()),
   781→   781→  );
   782→   782→
   783→   783→  return (
   784→   784→    <div className="space-y-2">
   785→   785→      <Label className="inline-flex items-center gap-1.5">
   786→   786→        <Icon className="h-3.5 w-3.5 text-muted-foreground" /> {label}
   787→   787→      </Label>
   788→   788→
   789→   789→      <div className="flex gap-2">
   790→   790→        <Input
   791→   791→          value={input}
   792→   792→          onChange={(e) => setInput(e.target.value)}
   793→   793→          onKeyDown={handleKeyDown}
   794→   794→          placeholder={placeholder}
   795→   795→        />
   796→   796→        <Button
   797→   797→          type="button"
   798→   798→          variant="outline"
   799→   799→          onClick={() => add(input)}
   800→   800→          disabled={!input.trim()}
   801→   801→        >
   802→   802→          <Plus className="mr-1 h-3.5 w-3.5" /> Add
   803→   803→        </Button>
   804→   804→      </div>
   805→   805→
   806→   806→      {values.length > 0 && (
   807→   807→        <div className="flex flex-wrap gap-1.5">
   808→   808→          {values.map((v, i) => (
   809→   809→            <span
   810→   810→              key={`${v}-${i}`}
   811→   811→              className="inline-flex items-center gap-1 rounded-full border border-ink/15 bg-ink/5 py-1 pl-3 pr-1.5 text-xs font-medium text-ink"
   812→   812→            >
   813→   813→              {v}
   814→   814→              <button
   815→   815→                type="button"
   816→   816→                onClick={() => remove(i)}
   817→   817→                className="inline-flex h-4 w-4 items-center justify-center rounded-full text-foreground/50 hover:bg-foreground/10 hover:text-foreground"
   818→   818→                aria-label={`Remove ${v}`}
   819→   819→              >
   820→   820→                <X className="h-3 w-3" />
   821→   821→              </button>
   822→   822→            </span>
   823→   823→          ))}
   824→   824→        </div>
   825→   825→      )}
   826→   826→
   827→   827→      {remainingSuggestions.length > 0 && (
   828→   828→        <div className="flex flex-wrap items-center gap-1.5 pt-1">
   829→   829→          <span className="inline-flex items-center gap-1 text-[11px] uppercase tracking-wide text-muted-foreground">
   830→   830→            <Sparkles className="h-3 w-3 text-sage" /> Suggested
   831→   831→          </span>
   832→   832→          {remainingSuggestions.map((s) => (
   833→   833→            <button
   834→   834→              key={s}
   835→   835→              type="button"
   836→   836→              onClick={() => add(s)}
   837→   837→              className="inline-flex items-center gap-1 rounded-full border border-sage/30 bg-sage/10 px-2.5 py-1 text-[11px] font-medium text-sage hover:bg-sage/20"
   838→   838→            >
   839→   839→              <Plus className="h-2.5 w-2.5" /> {s}
   840→   840→            </button>
   841→   841→          ))}
   842→   842→        </div>
   843→   843→      )}
   844→   844→    </div>
   845→   845→  );
   846→   846→}
