    68→
    69→export function BusinessProfileView() {
    70→  const route = useApp((s) => s.route);
    71→  if (route.name !== "business") return null;
    72→  // `key` ensures the inner component fully remounts on slug change → all
    73→  // state (loading, notFound, reviews, form) resets cleanly between profiles.
    74→  return <BusinessProfile key={route.slug} slug={route.slug} />;
    75→}
    76→
    77→function BusinessProfile({ slug }: { slug: string }) {
    78→  const navigate = useApp((s) => s.navigate);
    79→  const authUser = useApp((s) => s.authUser);
    80→  const { toast } = useToast();
    81→
    82→  const [business, setBusiness] = useState<ProfileBusiness | null>(null);
    83→  const [loading, setLoading] = useState(true);
    84→  const [notFound, setNotFound] = useState(false);
    85→
    86→  // review form
    87→  const [showReviewForm, setShowReviewForm] = useState(false);
    88→  const [reviewRating, setReviewRating] = useState(5);
    89→  const [reviewText, setReviewText] = useState("");
    90→  const [reviewCompany, setReviewCompany] = useState("");
    91→  const [submittingReview, setSubmittingReview] = useState(false);
    92→  const [localReviews, setLocalReviews] = useState<BusinessReview[] | null>(null);
    93→
    94→  // ---------- fetch business + fire view increment ----------
    95→  useEffect(() => {
    96→    let cancelled = false;
    97→    api<ProfileResponse>(`/api/businesses/${encodeURIComponent(slug)}`)
    98→      .then((d) => {
    99→        if (cancelled) return;
   100→        setBusiness(d.business);
   101→        // fire-and-forget view increment
   102→        fetch(`/api/businesses/${encodeURIComponent(slug)}/view`, {
   103→          method: "POST",
   104→        }).catch(() => {
   105→          /* noop */
   106→        });
   107→      })
   108→      .catch((err: Error & { status?: number }) => {
   109→        if (cancelled) return;
   110→        if (err.status === 404 || err.status === 400) {
   111→          setNotFound(true);
   112→        } else {
   113→          setNotFound(true);
   114→        }
   115→      })
   116→      .finally(() => {
   117→        if (!cancelled) setLoading(false);
   118→      });
   119→    return () => {
   120→      cancelled = true;
   121→    };
   122→  }, [slug]);
   123→
   124→  // ---------- handlers ----------
   125→  const reviews: BusinessReview[] = useMemo(() => {
   126→    if (localReviews) return localReviews;
   127→    return business?.reviews ?? [];
   128→  }, [business, localReviews]);
   129→
   130→  const submitReview = async () => {
   131→    if (!reviewText.trim()) {
   132→      toast({ title: "Review text is required", description: "Please share a few words about your experience." });
   133→      return;
   134→    }
   135→    setSubmittingReview(true);
   136→    try {
   137→      const data = await api<ReviewPayload>(
   138→        `/api/businesses/${encodeURIComponent(slug)}/reviews`,
   139→        { method: "POST", json: { rating: reviewRating, review: reviewText.trim(), reviewerCompany: reviewCompany.trim() || undefined } },
   140→      );
   141→      setLocalReviews((prev) => [data.review, ...(prev ?? business?.reviews ?? [])]);
   142→      setReviewText("");
   143→      setReviewCompany("");
   144→      setReviewRating(5);
   145→      setShowReviewForm(false);
   146→      toast({ title: "Review posted", description: "Thanks for sharing your experience." });
   147→    } catch (err) {
   148→      const msg = err instanceof Error ? err.message : "Could not submit review.";
   149→      toast({ title: "Could not submit review", description: msg });
   150→    } finally {
   151→      setSubmittingReview(false);
   152→    }
   153→  };
   154→
   155→  const shareProfile = async () => {
   156→    const url = typeof window !== "undefined" ? `${window.location.origin}/#/business/${slug}` : "";
   157→    try {
   158→      if (navigator.share) {
   159→        await navigator.share({ title: business?.name ?? "BlakNet business", url });
   160→      } else if (navigator.clipboard) {
   161→        await navigator.clipboard.writeText(url);
   162→        toast({ title: "Link copied", description: "Share this business with your network." });
   163→      }
   164→    } catch {
   165→      /* user dismissed share — noop */
   166→    }
   167→  };
   168→
   169→  const searchByTag = (term: string) => {
   170→    if (typeof window !== "undefined") {
   171→      window.sessionStorage.setItem(SEARCH_KEY, term);
   172→    }
   173→    navigate({ name: "directory" });
   174→  };
   175→
   176→  // ---------- not found ----------
   177→  if (notFound && !loading) {
   178→    return (
   179→      <div className="mx-auto max-w-3xl px-4 py-24">
   180→        <EmptyState
   181→          icon={Frown}
   182→          title="Business not found"
   183→          description="The business you're looking for may have been removed or the link is incorrect."
   184→          action={
   185→            <Button variant="outline" onClick={() => navigate({ name: "directory" })}>
   186→              <ArrowLeft className="mr-1.5 h-4 w-4" /> Back to directory
   187→            </Button>
   188→          }
   189→        />
   190→      </div>
   191→    );
   192→  }
   193→
   194→  // ---------- loading skeleton ----------
   195→  if (loading || !business) {
   196→    return <ProfileSkeleton />;
   197→  }
   198→
   199→  const b = business;
   200→  const yearFounded = b.yearFounded;
   201→  const yearsInBusiness = yearFounded ? Math.max(0, new Date().getFullYear() - yearFounded) : null;
   202→  const websiteHref = b.website ? (b.website.startsWith("http") ? b.website : `https://${b.website}`) : null;
   203→  const whatsappHref = b.whatsapp ? `https://wa.me/${b.whatsapp.replace(/[^0-9]/g, "")}` : null;
   204→
   205→  return (
   206→    <div className="flex flex-col">
   207→      {/* ===== HEADER BAND ===== */}
   208→      <section className="bg-ink-grain text-cream">
   209→        <div className="mx-auto max-w-7xl px-4 py-10 sm:px-6 lg:px-8 lg:py-14">
   210→          <button
   211→            type="button"
   212→            onClick={() => navigate({ name: "directory" })}
   213→            className="mb-6 inline-flex items-center gap-1.5 text-sm text-cream/60 transition-colors hover:text-cream"
   214→          >
   215→            <ArrowLeft className="h-4 w-4" /> Directory
   216→          </button>
   217→
   218→          <div className="flex flex-col gap-6 lg:flex-row lg:items-start lg:justify-between">
   219→            <div className="flex flex-col gap-5 sm:flex-row sm:items-start">
   220→              {/* logo */}
   221→              <div className="h-20 w-20 shrink-0 overflow-hidden rounded-xl border border-cream/15 bg-cream/5">
   222→                {b.logoUrl ? (
   223→                  <img src={b.logoUrl} alt={b.name} className="h-full w-full object-cover" />
   224→                ) : (
   225→                  <div className="flex h-full w-full items-center justify-center font-display text-3xl text-cream">
   226→                    {b.name[0]}
   227→                  </div>
   228→                )}
   229→              </div>
   230→
   231→              <div className="min-w-0">
   232→                <h1 className="font-display text-3xl leading-tight tracking-tight sm:text-4xl md:text-[2.6rem]">
   233→                  {b.name}
   234→                </h1>
   235→                {b.tagline && (
   236→                  <p className="mt-1.5 max-w-xl text-cream/70 sm:text-lg">{b.tagline}</p>
   237→                )}
   238→
   239→                <div className="mt-4 flex flex-wrap items-center gap-2">
   240→                  {b.verificationStatus === "VERIFIED" && <VerifiedBadge status="VERIFIED" size="md" />}
   241→                  {b.verificationStatus === "PENDING" && <VerifiedBadge status="PENDING" size="md" />}
   242→                  {b.bbbeeLevel && <BBBEEBadge level={b.bbbeeLevel} />}
   243→                  {b.featured && <Pill tone="cream">Featured</Pill>}
   244→                  {b.industry?.name && (
   245→                    <Pill tone="cream">
   246→                      <Building2 className="h-3 w-3" />
   247→                      {b.industry.name}
   248→                    </Pill>
   249→                  )}
   250→                  {(b.city || b.province) && (
   251→                    <Pill tone="cream">
   252→                      <MapPin className="h-3 w-3" />
   253→                      {provinceCity(b)}
   254→                    </Pill>
   255→                  )}
   256→                </div>
   257→              </div>
   258→            </div>
   259→
   260→            {/* CTA stack (desktop) */}
   261→            <div className="flex flex-col gap-2 lg:items-end">
   262→              <div className="flex flex-wrap gap-2 lg:justify-end">
   263→                {websiteHref && (
   264→                  <Button asChild className="btn-lift h-10 bg-cream px-4 text-ink shadow-lg hover:bg-cream/90">
   265→                    <a href={websiteHref} target="_blank" rel="noreferrer noopener">
   266→                      <Globe className="h-4 w-4" /> Website
   267→                      <ExternalLink className="h-3 w-3 opacity-60" />
   268→                    </a>
   269→                  </Button>
   270→                )}
   271→                {b.email && (
   272→                  <Button asChild variant="outline" className="h-10 border-cream/25 bg-transparent px-4 text-cream hover:bg-cream/10 hover:text-cream">
   273→                    <a href={`mailto:${b.email}`}>
   274→                      <Mail className="h-4 w-4" /> Email
   275→                    </a>
   276→                  </Button>
   277→                )}
   278→                {whatsappHref && (
   279→                  <Button asChild className={cn(
   280→                    "btn-lift h-10 px-4",
   281→                    websiteHref
   282→                      ? "border border-sage/40 bg-sage/10 text-cream hover:bg-sage/20"
   283→                      : "bg-sage text-cream shadow-lg hover:bg-sage/90",
   284→                  )}>
   285→                    <a href={whatsappHref} target="_blank" rel="noreferrer noopener">
   286→                      <MessageCircle className="h-4 w-4" /> WhatsApp
   287→                    </a>
   288→                  </Button>
   289→                )}
   290→                {b.isOwner ? (
   291→                  <span className="inline-flex h-10 items-center gap-1.5 rounded-md border border-cream/20 bg-cream/5 px-4 text-sm font-medium text-cream/80">
   292→                    <Building2 className="h-4 w-4" /> Your business
   293→                  </span>
   294→                ) : (
   295→                  <FollowButton
   296→                    businessId={b.id}
   297→                    businessName={b.name}
   298→                    initialFollowing={b.following ?? false}
   299→                    followerCount={b.followerCount ?? 0}
   300→                    tone="onDark"
   301→                  />
   302→                )}
   303→                <Button variant="ghost" onClick={shareProfile} className="h-10 px-4 text-cream/70 hover:bg-cream/10 hover:text-cream">
   304→                  <Share2 className="h-4 w-4" />
   305→                  <span className="hidden sm:inline">Share</span>
   306→                </Button>
   307→              </div>
   308→            </div>
   309→          </div>
   310→        </div>
   311→      </section>
   312→
   313→      {/* ===== PROFILE COMPLETION BAR (owner only) ===== */}
   314→      {b.isOwner && (
   315→        <section className="border-b border-border bg-cream-grain">
   316→          <div className="mx-auto max-w-7xl px-4 py-4 sm:px-6 lg:px-8">
   317→            <div className="flex flex-col gap-3 rounded-xl border border-border bg-card p-4 sm:flex-row sm:items-center sm:justify-between">
   318→              <div className="flex-1">
   319→                <div className="flex items-center justify-between gap-3">
   320→                  <p className="text-sm font-medium">
   321→                    Your profile is {b.profileCompletion}% complete
   322→                  </p>
   323→                  <span className="text-xs text-muted-foreground">
   324→                    {b.profileCompletion >= 80 ? "Looking great!" : "Add more details to rank higher in search."}
   325→                  </span>
   326→                </div>
   327→                <Progress value={b.profileCompletion} className="mt-2 h-1.5" />
   328→              </div>
   329→              <Button
   330→                variant="outline"
   331→                size="sm"
   332→                onClick={() => navigate({ name: "dashboard-business", id: b.id })}
   333→              >
   334→                Edit business
   335→              </Button>
   336→            </div>
   337→          </div>
   338→        </section>
   339→      )}
   340→
   341→      {/* ===== BODY ===== */}
   342→      <section className="bg-background">
   343→        <div className="mx-auto max-w-7xl px-4 pb-10 pt-2 sm:px-6 sm:pt-4 lg:px-8 lg:pb-10">
   344→          <div className="grid gap-8 lg:grid-cols-[minmax(0,2fr)_minmax(0,1fr)]">
   345→            {/* ===== MAIN COLUMN ===== */}
   346→            <div className="flex flex-col gap-6">
   347→              {/* About — first content card, overlaps the dark header for a layered "rising out of the dark" effect */}
   348→              <Card className="-mt-4 sm:-mt-8 relative z-10 shadow-xl">
   349→                <CardTitle>About</CardTitle>
   350→                <Separator className="mb-4" />
   351→                {b.description ? (
   352→                  <div className="space-y-3 text-sm leading-relaxed text-foreground/80">
   353→                    {b.description.split(/\n{2,}|\r\n{2,}/).map((para, i) => (
   354→                      <p key={i}>{para}</p>
   355→                    ))}
   356→                  </div>
   357→                ) : (
   358→                  <p className="text-sm italic text-muted-foreground">
   359→                    This business hasn't added a description yet.
   360→                  </p>
   361→                )}
   362→              </Card>
   363→
   364→              {/* Services & Products — single card with two clearly-titled subsections; pills are clickable (navigate to directory search) */}
   365→              {((b.services && b.services.length > 0) || (b.products && b.products.length > 0)) && (
   366→                <Card>
   367→                  <CardTitle>Services & Products</CardTitle>
   368→                  <Separator className="mb-4" />
   369→
   370→                  {b.services && b.services.length > 0 && (
   371→                    <div className={b.products && b.products.length > 0 ? "mb-6" : ""}>
   372→                      <h3 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   373→                        Services
   374→                      </h3>
   375→                      <div className="flex flex-wrap gap-2">
   376→                        {b.services.map((s) => (
   377→                          <button
   378→                            key={s.id}
   379→                            type="button"
   380→                            onClick={() => searchByTag(s.name)}
   381→                            className="transition-transform hover:-translate-y-0.5"
   382→                          >
   383→                            <Pill
   384→                              tone="neutral"
   385→                              className="cursor-pointer transition-colors hover:bg-ink hover:text-cream hover:border-ink"
   386→                            >
   387→                              {s.name}
   388→                            </Pill>
   389→                          </button>
   390→                        ))}
   391→                      </div>
   392→                    </div>
   393→                  )}
   394→
   395→                  {b.products && b.products.length > 0 && (
   396→                    <div>
   397→                      <h3 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   398→                        Products
   399→                      </h3>
   400→                      <div className="flex flex-wrap gap-2">
   401→                        {b.products.map((p) => (
   402→                          <button
   403→                            key={p.id}
   404→                            type="button"
   405→                            onClick={() => searchByTag(p.name)}
   406→                            className="transition-transform hover:-translate-y-0.5"
   407→                          >
   408→                            <Pill
   409→                              tone="sage"
   410→                              className="cursor-pointer transition-colors hover:bg-ink hover:text-cream hover:border-ink"
   411→                            >
   412→                              {p.name}
   413→                            </Pill>
   414→                          </button>
   415→                        ))}
   416→                      </div>
   417→                    </div>
   418→                  )}
   419→                </Card>
   420→              )}
   421→
   422→              {/* Reviews */}
   423→              <Card>
   424→                {/* Header: rating summary + Leave-a-review CTA */}
   425→                <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">

   428→                      {b.rating > 0 ? b.rating.toFixed(1) : "—"}

   455→                </div>
   456→                <Separator className="mb-4 mt-4" />

   458→                {/* inline review form */}

   785→function sizeLabel(size: string | null): string | null {
   786→  if (!size) return null;
   787→  const found = (
   788→    [
   789→      { value: "micro", label: "Micro (1–10)" },
   790→      { value: "small", label: "Small (11–50)" },
   791→      { value: "medium", label: "Medium (51–200)" },
   792→      { value: "large", label: "Large (200+)" },
   793→    ] as const
   794→  ).find((s) => s.value === size);
   795→  return found ? found.label : size.charAt(0).toUpperCase() + size.slice(1);
   796→}

   798→function prettyDomain(url: string): string {
   799→  try {
   800→    const u = new URL(url);
   801→    return u.hostname.replace(/^www\./, "");
   802→  } catch {
   803→    return url;
   804→  }
