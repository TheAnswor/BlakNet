    44→// ---------- types ----------
    45→type ProfileBusiness = Business & {

    61→const SEARCH_KEY = "blaknet:directory-search";
    62→
    63→export function BusinessProfileView() {
    64→  const route = useApp((s) => s.route);
    65→  if (route.name !== "business") return null;
    66→  // `key` ensures the inner component fully remounts on slug change → all
    67→  // state (loading, notFound, reviews, form) resets cleanly between profiles.
    68→  return <BusinessProfile key={route.slug} slug={route.slug} />;
    69→}
    70→
    71→function BusinessProfile({ slug }: { slug: string }) {
    72→  const navigate = useApp((s) => s.navigate);
    73→  const authUser = useApp((s) => s.authUser);
    74→  const { toast } = useToast();
    75→
    76→  const [business, setBusiness] = useState<ProfileBusiness | null>(null);
    77→  const [loading, setLoading] = useState(true);
    78→  const [notFound, setNotFound] = useState(false);
    79→
    80→  // review form
    81→  const [showReviewForm, setShowReviewForm] = useState(false);
    82→  const [reviewRating, setReviewRating] = useState(5);
    83→  const [reviewText, setReviewText] = useState("");
    84→  const [reviewCompany, setReviewCompany] = useState("");
    85→  const [submittingReview, setSubmittingReview] = useState(false);
    86→  const [localReviews, setLocalReviews] = useState<BusinessReview[] | null>(null);
    87→
    88→  // ---------- fetch business + fire view increment ----------
    89→  useEffect(() => {
    90→    let cancelled = false;
    91→    api<ProfileResponse>(`/api/businesses/${encodeURIComponent(slug)}`)
    92→      .then((d) => {
    93→        if (cancelled) return;
    94→        setBusiness(d.business);
    95→        // fire-and-forget view increment
    96→        fetch(`/api/businesses/${encodeURIComponent(slug)}/view`, {
    97→          method: "POST",
    98→        }).catch(() => {
    99→          /* noop */
   100→        });
   101→      })
   102→      .catch((err: Error & { status?: number }) => {
   103→        if (cancelled) return;
   104→        if (err.status === 404 || err.status === 400) {
   105→          setNotFound(true);
   106→        } else {
   107→          setNotFound(true);
   108→        }
   109→      })
   110→      .finally(() => {
   111→        if (!cancelled) setLoading(false);
   112→      });
   113→    return () => {
   114→      cancelled = true;
   115→    };
   116→  }, [slug]);
   117→
   118→  // ---------- handlers ----------
   119→  const reviews: BusinessReview[] = useMemo(() => {
   120→    if (localReviews) return localReviews;
   121→    return business?.reviews ?? [];
   122→  }, [business, localReviews]);
   123→
   124→  const submitReview = async () => {
   125→    if (!reviewText.trim()) {
   126→      toast({ title: "Review text is required", description: "Please share a few words about your experience." });
   127→      return;
   128→    }
   129→    setSubmittingReview(true);
   130→    try {
   131→      const data = await api<ReviewPayload>(
   132→        `/api/businesses/${encodeURIComponent(slug)}/reviews`,
   133→        { method: "POST", json: { rating: reviewRating, review: reviewText.trim(), reviewerCompany: reviewCompany.trim() || undefined } },
   134→      );
   135→      setLocalReviews((prev) => [data.review, ...(prev ?? business?.reviews ?? [])]);
   136→      setReviewText("");
   137→      setReviewCompany("");
   138→      setReviewRating(5);
   139→      setShowReviewForm(false);
   140→      toast({ title: "Review posted", description: "Thanks for sharing your experience." });
   141→    } catch (err) {
   142→      const msg = err instanceof Error ? err.message : "Could not submit review.";
   143→      toast({ title: "Could not submit review", description: msg });
   144→    } finally {
   145→      setSubmittingReview(false);
   146→    }
   147→  };
   148→
   149→  const shareProfile = async () => {
   150→    const url = typeof window !== "undefined" ? `${window.location.origin}/#/business/${slug}` : "";
   151→    try {
   152→      if (navigator.share) {
   153→        await navigator.share({ title: business?.name ?? "BlakNet business", url });
   154→      } else if (navigator.clipboard) {
   155→        await navigator.clipboard.writeText(url);
   156→        toast({ title: "Link copied", description: "Share this business with your network." });
   157→      }
   158→    } catch {
   159→      /* user dismissed share — noop */
   160→    }
   161→  };
   162→
   163→  const searchByTag = (term: string) => {
   164→    if (typeof window !== "undefined") {
   165→      window.sessionStorage.setItem(SEARCH_KEY, term);
   166→    }
   167→    navigate({ name: "directory" });
   168→  };
   169→
   170→  // ---------- not found ----------
   171→  if (notFound && !loading) {
   172→    return (
   173→      <div className="mx-auto max-w-3xl px-4 py-24">
   174→        <EmptyState
   175→          icon={Frown}
   176→          title="Business not found"
   177→          description="The business you're looking for may have been removed or the link is incorrect."
   178→          action={
   179→            <Button variant="outline" onClick={() => navigate({ name: "directory" })}>
   180→              <ArrowLeft className="mr-1.5 h-4 w-4" /> Back to directory
   181→            </Button>
   182→          }
   183→        />
   184→      </div>
   185→    );
   186→  }
   187→
   188→  // ---------- loading skeleton ----------
   189→  if (loading || !business) {
   190→    return <ProfileSkeleton />;
   191→  }
   192→
   193→  const b = business;
   194→  const yearFounded = b.yearFounded;
   195→  const yearsInBusiness = yearFounded ? Math.max(0, new Date().getFullYear() - yearFounded) : null;
   196→  const websiteHref = b.website ? (b.website.startsWith("http") ? b.website : `https://${b.website}`) : null;
   197→  const whatsappHref = b.whatsapp ? `https://wa.me/${b.whatsapp.replace(/[^0-9]/g, "")}` : null;
   198→
   199→  return (
   200→    <div className="flex flex-col">
   201→      {/* ===== HEADER BAND ===== */}
   202→      <section className="bg-ink-grain text-cream">
   203→        <div className="mx-auto max-w-7xl px-4 py-10 sm:px-6 lg:px-8 lg:py-14">
   204→          <button
   205→            type="button"
   206→            onClick={() => navigate({ name: "directory" })}
   207→            className="mb-6 inline-flex items-center gap-1.5 text-sm text-cream/60 transition-colors hover:text-cream"
   208→          >
   209→            <ArrowLeft className="h-4 w-4" /> Directory
   210→          </button>
   211→
   212→          <div className="flex flex-col gap-6 lg:flex-row lg:items-start lg:justify-between">
   213→            <div className="flex flex-col gap-5 sm:flex-row sm:items-start">
   214→              {/* logo */}
   215→              <div className="h-20 w-20 shrink-0 overflow-hidden rounded-xl border border-cream/15 bg-cream/5">
   216→                {b.logoUrl ? (
   217→                  <img src={b.logoUrl} alt={b.name} className="h-full w-full object-cover" />
   218→                ) : (
   219→                  <div className="flex h-full w-full items-center justify-center font-display text-3xl text-cream">
   220→                    {b.name[0]}
   221→                  </div>
   222→                )}
   223→              </div>
   224→
   225→              <div className="min-w-0">
   226→                <h1 className="font-display text-3xl leading-tight tracking-tight sm:text-4xl md:text-[2.6rem]">
   227→                  {b.name}
   228→                </h1>
   229→                {b.tagline && (
   230→                  <p className="mt-1.5 max-w-xl text-cream/70 sm:text-lg">{b.tagline}</p>
   231→                )}
   232→
   233→                <div className="mt-4 flex flex-wrap items-center gap-2">
   234→                  {b.verificationStatus === "VERIFIED" && <VerifiedBadge status="VERIFIED" size="md" />}
   235→                  {b.verificationStatus === "PENDING" && <VerifiedBadge status="PENDING" size="md" />}
   236→                  {b.bbbeeLevel && <BBBEEBadge level={b.bbbeeLevel} />}
   237→                  {b.featured && <Pill tone="cream">Featured</Pill>}
   238→                  {b.industry?.name && (
   239→                    <Pill tone="cream">
   240→                      <Building2 className="h-3 w-3" />
   241→                      {b.industry.name}
   242→                    </Pill>
   243→                  )}
   244→                  {(b.city || b.province) && (
   245→                    <Pill tone="cream">
   246→                      <MapPin className="h-3 w-3" />
   247→                      {provinceCity(b)}
   248→                    </Pill>
   249→                  )}
   250→                </div>
   251→              </div>
   252→            </div>
   253→
   254→            {/* CTA stack (desktop) */}
   255→            <div className="flex flex-col gap-2 lg:items-end">
   256→              <div className="flex flex-wrap gap-2 lg:justify-end">
   257→                {websiteHref && (
   258→                  <Button asChild className="bg-cream text-ink hover:bg-cream/90">
   259→                    <a href={websiteHref} target="_blank" rel="noreferrer noopener">
   260→                      <Globe className="h-4 w-4" /> Website
   261→                      <ExternalLink className="h-3 w-3 opacity-60" />
   262→                    </a>
   263→                  </Button>
   264→                )}
   265→                {b.email && (
   266→                  <Button asChild variant="outline" className="border-cream/20 bg-transparent text-cream hover:bg-cream/10 hover:text-cream">
   267→                    <a href={`mailto:${b.email}`}>
   268→                      <Mail className="h-4 w-4" /> Email
   269→                    </a>
   270→                  </Button>
   271→                )}
   272→                {whatsappHref && (
   273→                  <Button asChild className="bg-sage text-cream hover:bg-sage/90">
   274→                    <a href={whatsappHref} target="_blank" rel="noreferrer noopener">
   275→                      <MessageCircle className="h-4 w-4" /> WhatsApp
   276→                    </a>
   277→                  </Button>
   278→                )}
   279→                <Button variant="outline" onClick={shareProfile} className="border-cream/20 bg-transparent text-cream hover:bg-cream/10 hover:text-cream">
   280→                  <Share2 className="h-4 w-4" />
   281→                  <span className="hidden sm:inline">Share</span>
   282→                </Button>
   283→              </div>
   284→            </div>
   285→          </div>
   286→        </div>
   287→      </section>
   288→
   289→      {/* ===== PROFILE COMPLETION BAR (owner only) ===== */}
   290→      {b.isOwner && (
   291→        <section className="border-b border-border bg-cream-grain">
   292→          <div className="mx-auto max-w-7xl px-4 py-4 sm:px-6 lg:px-8">
   293→            <div className="flex flex-col gap-3 rounded-xl border border-border bg-card p-4 sm:flex-row sm:items-center sm:justify-between">
   294→              <div className="flex-1">
   295→                <div className="flex items-center justify-between gap-3">
   296→                  <p className="text-sm font-medium">
   297→                    Your profile is {b.profileCompletion}% complete
   298→                  </p>
   299→                  <span className="text-xs text-muted-foreground">
   300→                    {b.profileCompletion >= 80 ? "Looking great!" : "Add more details to rank higher in search."}
   301→                  </span>
   302→                </div>
   303→                <Progress value={b.profileCompletion} className="mt-2 h-1.5" />
   304→              </div>
   305→              <Button
   306→                variant="outline"
   307→                size="sm"
   308→                onClick={() => navigate({ name: "dashboard-business", id: b.id })}
   309→              >
   310→                Edit business
   311→              </Button>
   312→            </div>
   313→          </div>
   314→        </section>
   315→      )}
   316→
   317→      {/* ===== BODY ===== */}
   318→      <section className="bg-background">
   319→        <div className="mx-auto max-w-7xl px-4 py-10 sm:px-6 lg:px-8">
   320→          <div className="grid gap-8 lg:grid-cols-[minmax(0,2fr)_minmax(0,1fr)]">
   321→            {/* ===== MAIN COLUMN ===== */}
   322→            <div className="flex flex-col gap-6">
   323→              {/* About */}
   324→              <Card>
   325→                <CardTitle>About</CardTitle>
   326→                <Separator className="mb-4" />
   327→                {b.description ? (
   328→                  <div className="space-y-3 text-sm leading-relaxed text-foreground/80">
   329→                    {b.description.split(/\n{2,}|\r\n{2,}/).map((para, i) => (
   330→                      <p key={i}>{para}</p>
   331→                    ))}
   332→                  </div>
   333→                ) : (
   334→                  <p className="text-sm italic text-muted-foreground">
   335→                    This business hasn't added a description yet.
   336→                  </p>
   337→                )}
   338→              </Card>
   339→
   340→              {/* Services */}
   341→              {b.services && b.services.length > 0 && (
   342→                <Card>
   343→                  <CardTitle>Services</CardTitle>
   344→                  <Separator className="mb-4" />
   345→                  <div className="flex flex-wrap gap-2">
   346→                    {b.services.map((s) => (
   347→                      <button
   348→                        key={s.id}
   349→                        type="button"
   350→                        onClick={() => searchByTag(s.name)}
   351→                        className="transition-transform hover:-translate-y-0.5"
   352→                      >
   353→                        <Pill tone="neutral" className="cursor-pointer hover:border-foreground/40">
   354→                          {s.name}
   355→                        </Pill>
   356→                      </button>
   357→                    ))}
   358→                  </div>
   359→                </Card>
   360→              )}
   361→
   362→              {/* Products */}
   363→              {b.products && b.products.length > 0 && (
   364→                <Card>
   365→                  <CardTitle>Products</CardTitle>
   366→                  <Separator className="mb-4" />
   367→                  <div className="flex flex-wrap gap-2">
   368→                    {b.products.map((p) => (
   369→                      <button
   370→                        key={p.id}
   371→                        type="button"
   372→                        onClick={() => searchByTag(p.name)}
   373→                        className="transition-transform hover:-translate-y-0.5"
   374→                      >
   375→                        <Pill tone="sage" className="cursor-pointer hover:border-sage/60">
   376→                          {p.name}
   377→                        </Pill>
   378→                      </button>
   379→                    ))}
   380→                  </div>
   381→                </Card>
   382→              )}
   383→
   384→              {/* Reviews */}
   385→              <Card>
   386→                <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   387→                  <CardTitle className="mb-0">Reviews</CardTitle>
   388→                  {!showReviewForm && (
   389→                    <Button
   390→                      variant="outline"
   391→                      size="sm"
   392→                      onClick={() => {
   393→                        if (!authUser) {
   394→                          navigate({ name: "login" });
   395→                          return;
   396→                        }
   397→                        setShowReviewForm(true);
   398→                      }}
   399→                    >
   400→                      <Plus className="h-4 w-4" /> Leave a review

   403→                </div>
   404→                <Separator className="mb-4 mt-4" />
   405→
   406→                {/* summary */}
   407→                <div className="mb-5 flex flex-wrap items-center gap-4 rounded-lg border border-border bg-cream-grain/60 px-4 py-3">
   408→                  <div className="font-display text-3xl text-ink">
   409→                    {b.rating > 0 ? b.rating.toFixed(1) : "—"}
   410→                  </div>
   411→                  <div className="flex flex-col">
   412→                    <StarRating rating={b.rating} showValue={false} />
   413→                    <span className="mt-0.5 text-xs text-muted-foreground">
   414→                      Based on {formatNumber(b.reviewCount)} {b.reviewCount === 1 ? "review" : "reviews"}
   415→                    </span>
   416→                  </div>
   417→                </div>
   418→
   419→                {/* inline review form */}
   420→                {showReviewForm && (
   421→                  <div className="mb-5 rounded-lg border border-border bg-cream-grain/40 p-4">
   422→                    <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center">
   423→                      <label className="text-sm font-medium">Your rating</label>
   424→                      <Select value={String(reviewRating)} onValueChange={(v) => setReviewRating(Number(v))}>
   425→                        <SelectTrigger className="h-9 w-[160px]">
   426→                          <SelectValue />
   427→                        </SelectTrigger>
   428→                        <SelectContent>
   429→                          {[5, 4, 3, 2, 1].map((n) => (
   430→                            <SelectItem key={n} value={String(n)}>
   431→                              {n} {n === 1 ? "star" : "stars"}
   432→                            </SelectItem>

   437→                        <StarRating rating={reviewRating} />

   443→                      placeholder="Share your experience working with this business…"

   494→                      </li>

   496→                  </ul>
   497→                )}
   498→              </Card>
   499→            </div>
   500→
   501→            {/* ===== SIDEBAR ===== */}
   502→            <aside className="flex flex-col gap-6">

   517→
   518→                {/* Contact */}
   519→                <Card>
   520→                  <CardTitle>Contact</CardTitle>

   740→function sizeLabel(size: string | null): string | null {
   741→  if (!size) return null;
   742→  const found = (
   743→    [
   744→      { value: "micro", label: "Micro (1–10)" },
   745→      { value: "small", label: "Small (11–50)" },
   746→      { value: "medium", label: "Medium (51–200)" },
   747→      { value: "large", label: "Large (200+)" },
   748→    ] as const
   749→  ).find((s) => s.value === size);
   750→  return found ? found.label : size.charAt(0).toUpperCase() + size.slice(1);
   751→}
